# M1-C — HEVC + HDR Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add HEVC (H.265) encode + full HDR (HLG10/Main10) capture and encode, sent over enhanced-RTMP, with auto-best codec selection, a simple HDR toggle, and automatic fallback to H.264/SDR.

**Architecture:** A C++ `VideoCodec` strategy (`AvcCodec`/`HevcCodec`) in the FLV layer produces the FLV/enhanced-RTMP video payload; `RtmpClient` negotiates the codec at the enhanced connect handshake. Kotlin detects device caps, a pure `CodecSelector` picks the format, `VideoEncoder` configures MediaCodec for HEVC/Main10/HLG, and `Camera2Controller` outputs HLG10. The engine negotiates the codec **before** creating the encoder (encoder built at Live with the negotiated codec).

**Tech Stack:** C++17 NDK (GoogleTest), JNI, Kotlin, Camera2 (`DYNAMIC_RANGE_PROFILES`), MediaCodec (`video/hevc`, Main10, HLG/BT2020), enhanced-RTMP (Veovera).

**Spec:** `docs/superpowers/specs/2026-06-14-m1c-hevc-hdr-design.md`

**Enhanced-RTMP HEVC format (Veovera enhanced-RTMP v1):**
- ex-video tag first byte: `(IsExHeader=1)<<7 | (FrameType)<<4 | (PacketType)`. FrameType: key=1, inter=2. PacketType: SequenceStart=0, CodedFrames=1, SequenceEnd=2, CodedFramesX=3.
- Then 4-byte FourCC `"hvc1"` (ASCII).
- SequenceStart payload = `hvcC`. CodedFrames payload = `[3-byte cts][length-prefixed NALUs]` (CodedFramesX omits cts). Use CodedFrames (with cts).
- `hvcC` (HEVCDecoderConfigurationRecord) packs VPS+SPS+PPS. HEVC NAL header is 2 bytes; `nalType = (nal[0] >> 1) & 0x3F` (VPS=32, SPS=33, PPS=34).
- Connect advertises HEVC via a `fourCcList` strict-array (`["hvc1","avc1"]`) in the connect command object; the server signals support in the `_result` (its own `fourCcList`/capsEx). Pin the exact `_result` field during Task 4 against a live Twitch/SRT capture; the **fallback behavior is fixed** (no HEVC advertised by server ⇒ use AvcCodec).

`core/hevc.cpp`:

```cpp
#include "hevc.h"
namespace ps {

// Find Annex-B NAL ranges [start,end) (3- or 4-byte start codes). Local copy of the AVC
// scanner; HEVC parsing differs only in the NAL-type extraction (2-byte header).
static std::vector<std::pair<size_t,size_t>> FindNals(const Bytes& d) {
    std::vector<size_t> starts; size_t i = 0, n = d.size();
    while (i + 3 <= n) {
        bool sc4 = (i + 4 <= n && d[i]==0 && d[i+1]==0 && d[i+2]==0 && d[i+3]==1);
        bool sc3 = (d[i]==0 && d[i+1]==0 && d[i+2]==1);
        if (sc4) { starts.push_back(i + 4); i += 4; }
        else if (sc3) { starts.push_back(i + 3); i += 3; }
        else ++i;
    }
    std::vector<std::pair<size_t,size_t>> nals;
    for (size_t k = 0; k < starts.size(); ++k) {
        size_t s = starts[k];
        size_t e = (k + 1 < starts.size()) ? starts[k+1] : n;
        if (k + 1 < starts.size()) {
            bool sc4 = (e >= 4 && d[e-1]==1 && d[e-2]==0 && d[e-3]==0 && d[e-4]==0);
            e -= sc4 ? 4 : 3;
        }
        if (e > s) nals.push_back({s, e});
    }
    return nals;
}

void SplitHevcParams(const Bytes& csd, Bytes& vps, Bytes& sps, Bytes& pps) {
    for (auto& r : FindNals(csd)) {
        if (r.second <= r.first) continue;
        uint8_t type = (csd[r.first] >> 1) & 0x3f;   // HEVC: 2-byte NAL header
        Bytes nal(csd.begin() + r.first, csd.begin() + r.second);
        if (type == 32) vps = nal; else if (type == 33) sps = nal; else if (type == 34) pps = nal;
    }
}

static void PutParamArray(Bytes& b, uint8_t nalType, const Bytes& nal) {
    PutU8(b, nalType & 0x3f);          // array_completeness(0)|reserved(0)|NAL_unit_type
    PutU16BE(b, 1);                    // numNalus = 1
    PutU16BE(b, (uint16_t)nal.size());
    PutBytes(b, nal.data(), nal.size());
}

Bytes BuildHvcC(const Bytes& vps, const Bytes& sps, const Bytes& pps) {
    Bytes b;
    PutU8(b, 0x01);                    // configurationVersion
    // general_profile_space/tier/idc, profile_compatibility(4), constraint(6): copy from SPS if
    // available; otherwise conservative defaults (decoders re-read these from the SPS NAL anyway).
    PutU8(b, 0x01);                    // general_profile_space(0)+tier(0)+profile_idc(1=Main)
    PutU32BE(b, 0x60000000);           // general_profile_compatibility_flags (Main)
    // general_constraint_indicator_flags (6 bytes)
    PutU8(b, 0x90); PutU8(b, 0); PutU8(b, 0); PutU8(b, 0); PutU8(b, 0); PutU8(b, 0);
    PutU8(b, 0x5d);                    // general_level_idc (~Level 3.1; decoder reads real level from SPS)
    PutU16BE(b, 0xf000);               // reserved(4=1111)+min_spatial_segmentation_idc(12)
    PutU8(b, 0xfc);                    // reserved(6)+parallelismType(2)
    PutU8(b, 0xfd);                    // reserved(6)+chromaFormat(2=4:2:0)
    PutU8(b, 0xf8);                    // reserved(5)+bitDepthLumaMinus8(3=0)
    PutU8(b, 0xf8);                    // reserved(5)+bitDepthChromaMinus8(3=0)
    PutU16BE(b, 0);                    // avgFrameRate
    PutU8(b, 0x0f);                    // constantFrameRate(2)+numTemporalLayers(3)+temporalIdNested(1)+lengthSizeMinusOne(2=3)
    PutU8(b, 0x03);                    // numOfArrays = 3
    PutParamArray(b, 32, vps);
    PutParamArray(b, 33, sps);
    PutParamArray(b, 34, pps);
    return b;
}
}
```

> Note: `hvcC`'s profile/level/bit-depth fields are informational — HEVC decoders (and ffmpeg) re-parse them from the SPS NAL carried in the record, so conservative constants are safe for M1-C. (Refinement to parse the SPS for exact values is a later polish item.)

#pragma once
#include "byte_writer.h"
namespace ps {
class VideoCodec {
public:
    virtual ~VideoCodec() = default;
    virtual Bytes SequenceHeader(const Bytes& csd) = 0;                  // FLV video message payload
    virtual Bytes Frame(const Bytes& annexb, bool key, uint32_t cts) = 0;
    virtual const char* FourCc() const = 0;
};
class AvcCodec : public VideoCodec {
public:
    Bytes SequenceHeader(const Bytes& csd) override;
    Bytes Frame(const Bytes& annexb, bool key, uint32_t cts) override;
    const char* FourCc() const override { return "avc1"; }
};
class HevcCodec : public VideoCodec {
public:
    Bytes SequenceHeader(const Bytes& csd) override;
    Bytes Frame(const Bytes& annexb, bool key, uint32_t cts) override;
    const char* FourCc() const override { return "hvc1"; }
};
}

#include "video_codec.h"
#include "flv.h"
#include "hevc.h"
namespace ps {

Bytes AvcCodec::SequenceHeader(const Bytes& csd) {
    Bytes sps, pps; SplitSpsPps(csd, sps, pps);
    return FlvVideoSeqHeader(BuildAvcC(sps, pps));
}
Bytes AvcCodec::Frame(const Bytes& annexb, bool key, uint32_t cts) {
    return FlvVideoFrame(AnnexBToAvcc(annexb), key, cts);
}

static void PutFourCc(Bytes& b, const char* cc) { for (int i = 0; i < 4; ++i) PutU8(b, (uint8_t)cc[i]); }

Bytes HevcCodec::SequenceHeader(const Bytes& csd) {
    Bytes vps, sps, pps; SplitHevcParams(csd, vps, sps, pps);
    Bytes b;
    PutU8(b, 0x80 | 0x10 | 0x00);               // IsExHeader | key | SequenceStart
    PutFourCc(b, "hvc1");
    Bytes hvcc = BuildHvcC(vps, sps, pps);
    PutBytes(b, hvcc.data(), hvcc.size());
    return b;
}
Bytes HevcCodec::Frame(const Bytes& annexb, bool key, uint32_t cts) {
    Bytes b;
    PutU8(b, 0x80 | (key ? 0x10 : 0x20) | 0x01); // IsExHeader | frameType | CodedFrames
    PutFourCc(b, "hvc1");
    PutU24BE(b, cts);
    Bytes nalus = AnnexBToAvcc(annexb);          // 4-byte length-prefixed NALUs (same as AVC)
    PutBytes(b, nalus.data(), nalus.size());
    return b;
}
}

```kotlin
package com.example.plohoystream.media

import android.media.MediaCodecInfo
import android.media.MediaCodecList
import android.media.MediaFormat

/** Queries the device's HEVC encoder support. */
object CodecCapabilities {
    data class Hevc(val encoder: Boolean, val main10: Boolean)

    fun hevc(): Hevc {
        val list = MediaCodecList(MediaCodecList.REGULAR_CODECS)
        for (info in list.codecInfos) {
            if (!info.isEncoder) continue
            if (!info.supportedTypes.any { it.equals(MediaFormat.MIMETYPE_VIDEO_HEVC, ignoreCase = true) }) continue
            val caps = runCatching { info.getCapabilitiesForType(MediaFormat.MIMETYPE_VIDEO_HEVC) }.getOrNull() ?: continue
            val main10 = caps.profileLevels.any {
                it.profile == MediaCodecInfo.CodecProfileLevel.HEVCProfileMain10 ||
                    it.profile == MediaCodecInfo.CodecProfileLevel.HEVCProfileMain10HDR10
            }
            return Hevc(encoder = true, main10 = main10)
        }
        return Hevc(encoder = false, main10 = false)
    }
}
```
